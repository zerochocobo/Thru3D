cmake_minimum_required(VERSION 3.22)
project(PgsBitmapChecks LANGUAGES CXX)
add_executable(pgs_bitmap_checks pgs_bitmap_checks.cpp)
target_compile_features(pgs_bitmap_checks PRIVATE cxx_std_17)
target_include_directories(pgs_bitmap_checks PRIVATE ../../native/mpv/include)
if(MSVC)
    target_compile_options(pgs_bitmap_checks PRIVATE /EHsc /W4 /WX)
else()
    target_compile_options(pgs_bitmap_checks PRIVATE -Wall -Wextra -Werror)
endif()
